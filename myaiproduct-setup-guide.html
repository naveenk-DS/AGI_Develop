import os, json, subprocess
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.kdf.scrypt import Scrypt
from cryptography.hazmat.backends import default_backend

def get_machine_id() -> str:
    """Unique ID from this PC's hardware."""
    result = subprocess.check_output(
        "wmic csproduct get uuid /value",
        shell=True, encoding="utf-8"
    )
    return result.strip().replace("UUID=", "")

def derive_key(machine_id: str) -> bytes:
    """Derive AES-256 key from hardware ID."""
    kdf = Scrypt(
        salt=b"myaiproduct-salt-v1",
        length=32, n=2**14, r=8, p=1,
        backend=default_backend()
    )
    return kdf.derive(machine_id.encode())

# Derived once at startup
_KEY = derive_key(get_machine_id())

def encrypt(data: dict) -> bytes:
    aesgcm = AESGCM(_KEY)
    iv = os.urandom(12)
    ct = aesgcm.encrypt(iv, json.dumps(data).encode(), None)
    return iv + ct

def decrypt(payload: bytes) -> dict:
    aesgcm = AESGCM(_KEY)
    iv, ct = payload[:12], payload[12:]
    plain = aesgcm.decrypt(iv, ct, None)
    return json.loads(plain.decode())

import httpx, os
from loguru import logger
from dotenv import load_dotenv

load_dotenv("config/.env")
OLLAMA = os.getenv("OLLAMA_URL", "http://127.0.0.1:11434")

async def chat(prompt: str, model: str = "llama3",
              system: str = None) -> str:
    messages = []
    if system:
        messages.append({"role": "system", "content": system})
    messages.append({"role": "user", "content": prompt})

    async with httpx.AsyncClient(timeout=120) as client:
        r = await client.post(f"{OLLAMA}/api/chat",
            json={"model": model, "messages": messages, "stream": False})
        r.raise_for_status()
        return r.json()["message"]["content"]

async def embed(text: str) -> list[float]:
    async with httpx.AsyncClient(timeout=60) as client:
        r = await client.post(f"{OLLAMA}/api/embeddings",
            json={"model": "nomic-embed-text", "prompt": text})
        r.raise_for_status()
        return r.json()["embedding"]

from fastapi import FastAPI
from contextlib import asynccontextmanager
from loguru import logger
import sys, os
from app.database import engine, Base
from app import models  # import so Base knows all tables

# Log to file + terminal
logger.remove()
logger.add(sys.stdout, level="INFO")
logger.add("logs/app.log", rotation="10 MB",
           retention="7 days", level="INFO")

@asynccontextmanager
async def lifespan(app: FastAPI):
    # Create all DB tables on startup
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
    logger.info("✅ Database tables ready")
    logger.info("✅ MyAIProduct API started")
    yield
    await engine.dispose()
    logger.info("API shutdown")

app = FastAPI(
    title="MyAIProduct API",
    version="1.0.0",
    lifespan=lifespan
)

@app.get("/health")
async def health():
    return {"status": "ok", "version": "1.0.0"}

@app.post("/chat")
async def chat(body: dict):
    from app.services.ollama import chat as ollama_chat
    reply = await ollama_chat(body.get("message", ""))
    return {"reply": reply}
